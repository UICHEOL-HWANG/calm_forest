`js/mountain/cafe-shop.js`:

```js
import { cafeItem } from './cafe-catalog.js';
import { findItem } from '../cosmetics/catalog.js';

export const LOCK_TEXT = '🏁 깃발 12개 모두 지나 완주하면 열려요';

const price = n => `${n.toLocaleString()}🪙`;

export function cafeState(gs) {
  return {
    coins: gs.inventory.coins | 0,
    cos: gs.cosmetics,
    hut: gs.mountain?.hut || { sleds: ['basic'], sled: 'basic' },
    perfect: !!gs.mountain?.sled?.perfect,
  };
}

export function rowOf(item, st) {
  if (item.tab === 'sled') {
    if (st.hut.sled === item.id) return { kind: 'riding', label: '타는 중', disabled: true };
    if (st.hut.sleds.includes(item.id)) return { kind: 'ride', label: '타기', disabled: false };
    if (item.needs === 'perfect' && !st.perfect) return { kind: 'locked', label: LOCK_TEXT, disabled: true };
  }
  if (item.cos && st.cos.owned.includes(item.id)) return { kind: 'owned', label: '구매 완료', disabled: true };
  return { kind: 'buy', label: price(item.coins), disabled: false };
}

function grantOf(item) {
  if (item.tab === 'drink') return { type: 'drink', id: item.id, cup: item.cup };
  if (item.tab === 'sled') return { type: 'sled', id: item.id };
  if (item.cos) return { type: 'cos', id: item.id };
  return { type: item.store, id: item.id };
}

function applyGrant(st, item) {
  if (item.tab === 'sled') return { ...st, hut: { sleds: [...st.hut.sleds, item.id], sled: item.id } };
  if (item.cos) {
    const slot = findItem(item.id).slot;
    return { ...st, cos: { owned: [...st.cos.owned, item.id], equipped: { ...st.cos.equipped, [slot]: item.id } } };
  }
  return st;
}

export function buyCafe(st, id) {
  const item = cafeItem(id);
  if (!item) return { ok: false, why: 'unknown', st, grant: null };
  const row = rowOf(item, st);
  if (row.kind === 'locked') return { ok: false, why: 'locked', st, grant: null };
  if (row.kind !== 'buy') return { ok: false, why: 'owned', st, grant: null };
  if (st.coins < item.coins) return { ok: false, why: 'poor', st, grant: null };
  const next = applyGrant({ ...st, coins: st.coins - item.coins }, item);
  return { ok: true, why: '', st: next, grant: grantOf(item) };
}

export function pickSled(st, id) {
  if (!st.hut.sleds.includes(id)) return st;
  return { ...st, hut: { ...st.hut, sled: id } };
}
```

```js
export const SLED_SKIN_IDS = Object.freeze(['basic', 'sky', 'forest', 'starry']);

function restoreHut(h) {
  const list = isObj(h) && Array.isArray(h.sleds) ? h.sleds : [];
  const sleds = ['basic', ...SLED_SKIN_IDS.filter(id => id !== 'basic' && list.includes(id))];
  const sled = isObj(h) && sleds.includes(h.sled) ? h.sled : 'basic';
  return { sleds, sled };
}
```

```js
function part(THREE, [size, at, tilt, hex]) {
  const g = colored(THREE, new THREE.BoxGeometry(...size), hex);
  return g.applyMatrix4(new THREE.Matrix4().compose(
    new THREE.Vector3(...at),
    new THREE.Quaternion().setFromEuler(new THREE.Euler(tilt, 0, 0)),
    new THREE.Vector3(1, 1, 1),
  ));
}

export function buildSledGeometry(THREE, skin = 'basic') {
  const pal = SLED_SKINS[skin] || SLED_SKINS.basic;
  const parts = PARTS.map(([s, a, t, key]) => part(THREE, [s, a, t, pal[key]]));
  const stars = pal.stars ? STARS.map(([x, z]) => part(THREE, [[0.07, 0.012, 0.07], [x, 0.226, z], 0, pal.runner])) : [];
  return mergeColored(THREE, [...parts, ...stars]);
}

export function buildSledMesh(THREE, skin = 'basic') {
  const mesh = new THREE.Mesh(buildSledGeometry(THREE, skin), new THREE.MeshLambertMaterial({ vertexColors: true, flatShading: true }));
  mesh.rotation.order = 'YXZ';
  mesh.visible = false;
  return mesh;
}

export function setSledSkin(mesh, THREE, skin) {
  const old = mesh.geometry;
  mesh.geometry = buildSledGeometry(THREE, skin);
  old.dispose();
}
```

```js
export function clampSummit(p, clampLocalFn) {
  if (inYard(p)) return { x: p.x, z: p.z, onPath: false, seg: 0, k: 0, yard: true };
  const c = clampLocalFn(p);
```

```js
  { id: 'counterL', x: 1.0, z: -1.8, r: 0.6 },
  { id: 'counterM', x: 2.2, z: -1.8, r: 0.6 },

  { id: 'ladder', x: -0.3, z: -1.6, r: 0.4 },
].map(c => Object.freeze(c)));

function pushOnce(p) {
  return CAFE_COLLIDERS.reduce((q, c) => {
    const dx = q.x - c.x, dz = q.z - c.z, d = Math.hypot(dx, dz), min = c.r + PLAYER_R;
    if (d >= min) return q;
    if (d < 1e-6) return { x: c.x, z: c.z + min };
    return { x: c.x + (dx / d) * min, z: c.z + (dz / d) * min };
  }, p);
}

export function clampCafe(p) {
  const hx = ROOM.w / 2 - WALL_IN, hz = ROOM.d / 2 - WALL_IN;
  let q = { x: Math.max(-hx, Math.min(hx, p.x)), z: Math.max(-hz, Math.min(hz + 0.3, p.z)) };
  for (let i = 0; i < 4; i++) {
    const next = pushOnce(q);
    if (next === q) break;
    q = next;
  }
  return q.x === p.x && q.z === p.z ? p : q;
}

export function nearestSeat(p, taken = []) {
  const open = SEATS.filter(s => !taken.some(t => t.x === s.x && t.z === s.z));
  return open.reduce((a, s) => (Math.hypot(s.x - p.x, s.z - p.z) < Math.hypot(a.x - p.x, a.z - p.z) ? s : a), open[0]);
}
```

```js
export const COLD = Object.freeze({
  max: 100,
  rate: Object.freeze({ day: 0.6, dusk: 0.8, night: 1.2, snow: 1.2 }),
  shiverAt: 60, slowAt: 80, slowMult: 0.75,
  fireR: 3, fireRate: 5, cafeRate: 8, warmS: 180,
});

export const COLD_START = Object.freeze({ v: 0, warmT: 0 });

export function stepCold(s, dt, { key, nearFire, inCafe, frozen }) {
  if (frozen) return s;
  const drop = inCafe ? COLD.cafeRate : nearFire ? COLD.fireRate : 0;
  const rise = drop > 0 || s.warmT > 0 ? 0 : (COLD.rate[key] ?? COLD.rate.day);
  const v = Math.min(COLD.max, Math.max(0, s.v + (rise - drop) * dt));
  return { ...s, v, warmT: Math.max(0, s.warmT - dt) };
}

export function drinkWarm(s) {
  return { ...s, v: 0, warmT: COLD.warmS };
}

export function coldStage(v) {
  return v >= COLD.slowAt ? 'slow' : v >= COLD.shiverAt ? 'shiver' : 'ok';
}

export function coldSpeedMult(v) {
  return v >= COLD.slowAt ? COLD.slowMult : 1;
}

export function noticeCold(prev, next, seenHint) {
  return {
    hint: !seenHint && prev.v < COLD.shiverAt && next.v >= COLD.shiverAt,
    slow: prev.v < COLD.slowAt && next.v >= COLD.slowAt,
  };
}
```
